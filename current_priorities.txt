"use client"
import { useState, useMemo, useEffect, useRef } from 'react';
import { 
  Wallet, 
  Lightbulb, 
  TrendingUp, 
  Package, 
  Plus, 
  Minus, 
  X, 
  Trash2, 
  ArrowDownUp, 
  List, 
  LayoutGrid, 
  Search,
  AlertCircle,
  BarChart3
} from 'lucide-react';
import Image from 'next/image';
import type { PriorityFilterType } from '@/app/page';
import { useApp, type PriorityType } from '@/lib/context/AppContext';

const FILTER_IMG = "https://lh3.googleusercontent.com/aida-public/AB6AXuCT4JVy2QYtBYNeSK0tcgqaJ-yFmDtSkdhC6Bwsc_pfoSksrfT7AWr0dNTBjLITm4IyuZYX7kuqkVQXaItletFxRHAnIwE5AGN4y__0NOeuGoGUj6EO9EQ4092ZNAZn7ec7XsnXOCYLEswAFNN44918KeFUN67s4d_AB-WkFReHYLlgNwlfLncp5r-J0LxlgiNknQrCKOD9dXy8-nv5QnjAAoGpRuBGNhi0PtDnlEDqnBxexK5wIQVzHQ";

interface PrioritiesViewProps {
  initialPriority?: PriorityFilterType;
  onPriorityChange?: (p: PriorityFilterType) => void;
}

export default function PrioritiesView({ 
  initialPriority = 'Todas',
  onPriorityChange 
}: PrioritiesViewProps) {
  const { 
    priorityItems, 
    addPriorityItem, 
    removePriorityItem, 
    updatePriorityItemQty, 
    updatePriorityItemLevel,
    priorityStats 
  } = useApp();

  const [localPriority, setLocalPriority] = useState<PriorityFilterType | null>(null);
  // Se houver prioridade externa passada e o usuário não mudou localmente nesta tela, ou se foi atualizada
  const activeFilter = onPriorityChange ? initialPriority : (localPriority ?? initialPriority);
  const [searchQuery, setSearchQuery] = useState('');
  const [sortBy, setSortBy] = useState<string>('Mais Urgente');
  const [viewMode, setViewMode] = useState<'list' | 'cards'>('list');

  // Modal para adicionar item
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [newItemName, setNewItemName] = useState('');
  const [newItemCategory, setNewItemCategory] = useState('');
  const [newItemPrice, setNewItemPrice] = useState('');
  const [newItemPriority, setNewItemPriority] = useState<PriorityType>('Alta');
  const [nameError, setNameError] = useState(false);

  const [openPriorityMenuId, setOpenPriorityMenuId] = useState<number | null>(null);
  const priorityMenuRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    function handleClickOutside(event: MouseEvent) {
      if (priorityMenuRef.current && !priorityMenuRef.current.contains(event.target as Node)) {
        setOpenPriorityMenuId(null);
      }
    }
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  const handleFilterSelect = (priority: PriorityFilterType) => {
    setLocalPriority(priority);
    if (onPriorityChange) {
      onPriorityChange(priority);
    }
  };

  const handleAdd = () => {
    if (!newItemName.trim()) {
      setNameError(true);
      return;
    }
    setNameError(false);
    
    const numPrice = parseFloat(newItemPrice.replace(/[^\d.,]/g, '').replace(',', '.')) || 0;
    
    addPriorityItem({
      name: newItemName.trim(),
      category: newItemCategory.trim() || 'Geral',
      price: numPrice > 0 ? `R$ ${numPrice.toFixed(2).replace('.', ',')}` : 'R$ 0,00',
      numPrice,
      img: FILTER_IMG,
      qty: 1,
      priority: newItemPriority
    });

    setIsModalOpen(false);
    setNewItemName('');
    setNewItemCategory('');
    setNewItemPrice('');
    setNewItemPriority('Alta');
  };

  // Filtragem e ordenação dos itens
  const filteredItems = useMemo(() => {
    let result = [...priorityItems];

    // Filtro por prioridade
    if (activeFilter !== 'Todas') {
      result = result.filter(i => i.priority === activeFilter);
    }

    // Filtro por busca de texto
    if (searchQuery.trim()) {
      const q = searchQuery.toLowerCase().trim();
      result = result.filter(i => 
        i.name.toLowerCase().includes(q) || 
        (i.category && i.category.toLowerCase().includes(q))
      );
    }

    // Ordenação
    switch (sortBy) {
      case 'Maior Preço':
        result.sort((a, b) => b.numPrice - a.numPrice);
        break;
      case 'Menor Preço':
        result.sort((a, b) => a.numPrice - b.numPrice);
        break;
      case 'Nome (A-Z)':
        result.sort((a, b) => a.name.localeCompare(b.name));
        break;
      case 'Mais Urgente':
      default:
        const priorityOrder: Record<PriorityType, number> = { 'Alta': 1, 'Média': 2, 'Baixa': 3 };
        result.sort((a, b) => {
          if (priorityOrder[a.priority] !== priorityOrder[b.priority]) {
            return priorityOrder[a.priority] - priorityOrder[b.priority];
          }
          return a.qty - b.qty;
        });
        break;
    }

    return result;
  }, [priorityItems, activeFilter, searchQuery, sortBy]);

  // Itens agrupados por prioridade para a visualização em lista
  const groupedItems = useMemo(() => {
    return {
      'Alta': filteredItems.filter(i => i.priority === 'Alta'),
      'Média': filteredItems.filter(i => i.priority === 'Média'),
      'Baixa': filteredItems.filter(i => i.priority === 'Baixa'),
    };
  }, [filteredItems]);

  const priorityConfigs: Record<PriorityType, { label: string; badgeClass: string; dotClass: string; borderClass: string; bgLight: string; textClass: string }> = {
    'Alta': {
      label: 'Alta',
      badgeClass: 'bg-rose-100 dark:bg-rose-950/60 text-rose-800 dark:text-rose-300 border-rose-200 dark:border-rose-800/70',
      dotClass: 'bg-rose-500',
      borderClass: 'border-l-rose-500',
      bgLight: 'bg-rose-50/50 dark:bg-rose-950/30',
      textClass: 'text-rose-700 dark:text-rose-300'
    },
    'Média': {
      label: 'Média',
      badgeClass: 'bg-amber-100 dark:bg-amber-950/60 text-amber-800 dark:text-amber-300 border-amber-200 dark:border-amber-800/70',
      dotClass: 'bg-amber-500',
      borderClass: 'border-l-amber-500',
      bgLight: 'bg-amber-50/50 dark:bg-amber-950/30',
      textClass: 'text-amber-700 dark:text-amber-300'
    },
    'Baixa': {
      label: 'Baixa',
      badgeClass: 'bg-blue-100 dark:bg-blue-950/60 text-blue-800 dark:text-blue-300 border-blue-200 dark:border-blue-800/70',
      dotClass: 'bg-blue-500',
      borderClass: 'border-l-blue-500',
      bgLight: 'bg-blue-50/50 dark:bg-blue-950/30',
      textClass: 'text-blue-700 dark:text-blue-300'
    }
  };

  // Dados calculados para o Menu de Investimento Total dinâmico
  const activeLevelInvestment = useMemo(() => {
    if (activeFilter === 'Alta') {
      return {
        title: 'Investimento • Alta',
        levelCost: priorityStats.highCost,
        levelCount: priorityStats.highCount,
        levelUnits: priorityStats.highUnits,
        levelPercent: priorityStats.highPercent,
        otherCost: Math.max(0, priorityStats.totalCost - priorityStats.highCost),
        accentColor: 'bg-rose-500',
        textColor: 'text-rose-700 dark:text-rose-300',
        badgeBg: 'bg-rose-100 dark:bg-rose-950/60 text-rose-800 dark:text-rose-300 border-rose-200 dark:border-rose-800/60',
      };
    }
    if (activeFilter === 'Média') {
      return {
        title: 'Investimento • Média',
        levelCost: priorityStats.medCost,
        levelCount: priorityStats.medCount,
        levelUnits: priorityStats.medUnits,
        levelPercent: priorityStats.medPercent,
        otherCost: Math.max(0, priorityStats.totalCost - priorityStats.medCost),
        accentColor: 'bg-amber-500',
        textColor: 'text-amber-700 dark:text-amber-300',
        badgeBg: 'bg-amber-100 dark:bg-amber-950/60 text-amber-800 dark:text-amber-300 border-amber-200 dark:border-amber-800/60',
      };
    }
    if (activeFilter === 'Baixa') {
      return {
        title: 'Investimento • Baixa',
        levelCost: priorityStats.lowCost,
        levelCount: priorityStats.lowCount,
        levelUnits: priorityStats.lowUnits,
        levelPercent: priorityStats.lowPercent,
        otherCost: Math.max(0, priorityStats.totalCost - priorityStats.lowCost),
        accentColor: 'bg-blue-500',
        textColor: 'text-blue-700 dark:text-blue-300',
        badgeBg: 'bg-blue-100 dark:bg-blue-950/60 text-blue-800 dark:text-blue-300 border-blue-200 dark:border-blue-800/60',
      };
    }
    return null;
  }, [activeFilter, priorityStats]);

  return (
    <div className="flex flex-col gap-6 animate-in fade-in slide-in-from-bottom-4 duration-500">
      {/* Cabeçalho */}
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
        <div>
          <h1 className="text-2xl md:text-3xl font-bold tracking-tight text-slate-900 dark:text-slate-100">Prioridades</h1>
          <p className="mt-1 text-slate-500 dark:text-slate-400">Organize itens e investimentos por urgência.</p>
        </div>
        <div className="flex items-center gap-3 w-full sm:w-auto">
          {/* Seletor de Modo de Exibição */}
          <div className="flex items-center bg-slate-100 dark:bg-slate-800 p-1 rounded-xl border border-slate-200 dark:border-slate-700">
            <button
              onClick={() => setViewMode('list')}
              className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-semibold transition-all ${
                viewMode === 'list'
                  ? 'bg-white dark:bg-slate-700 text-slate-900 dark:text-slate-100 shadow-xs'
                  : 'text-slate-600 dark:text-slate-400 hover:text-slate-900 dark:hover:text-slate-200'
              }`}
              title="Visualização em Lista por Prioridade"
            >
              <List className="w-4 h-4" />
              <span>Lista</span>
            </button>
            <button
              onClick={() => setViewMode('cards')}
              className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-semibold transition-all ${
                viewMode === 'cards'
                  ? 'bg-white dark:bg-slate-700 text-slate-900 dark:text-slate-100 shadow-xs'
                  : 'text-slate-600 dark:text-slate-400 hover:text-slate-900 dark:hover:text-slate-200'
              }`}
              title="Visualização em Cartões"
            >
              <LayoutGrid className="w-4 h-4" />
              <span>Cartões</span>
            </button>
          </div>

          <button 
            onClick={() => setIsModalOpen(true)}
            className="flex-1 sm:flex-none bg-slate-900 dark:bg-emerald-600 text-white px-4 py-2 rounded-xl text-sm font-semibold flex items-center justify-center gap-2 hover:bg-slate-800 dark:hover:bg-emerald-500 transition-colors shadow-sm active:scale-95"
          >
            <Plus className="h-4 w-4" /> Adicionar Item
          </button>
        </div>
      </div>

      {/* MENU DE INVESTIMENTO TOTAL DINÂMICO & CONTRASTANTE */}
      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 md:gap-6">
        {/* Card Interativo: Menu de Investimento por Nível vs Total Geral */}
        <div className="lg:col-span-2 bg-white dark:bg-slate-900 rounded-2xl p-6 border border-slate-200 dark:border-slate-800 shadow-sm flex flex-col justify-between relative overflow-hidden transition-colors">
          {/* Topo do Menu: Título e Seletor Inline de Prioridades */}
          <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-3 mb-4">
            <div className="flex items-center gap-2">
              <div className="bg-slate-900 dark:bg-slate-800 p-2 rounded-xl text-white">
                <Wallet className="h-4 w-4" />
              </div>
              <div>
                <h3 className="text-xs font-bold text-slate-400 dark:text-slate-500 uppercase tracking-wider">
                  Investimento
                </h3>
                <h2 className="text-base font-bold text-slate-900 dark:text-slate-100">
                  {activeLevelInvestment ? activeLevelInvestment.title : 'Total Geral'}
                </h2>
              </div>
            </div>

            {/* Submenu de troca rápida de prioridade diretamente dentro do card */}
            <div className="flex items-center gap-1 bg-slate-100 dark:bg-slate-800 p-1 rounded-xl border border-slate-200 dark:border-slate-700 w-full sm:w-auto overflow-x-auto">
              <button
                onClick={() => handleFilterSelect('Todas')}
                className={`px-2.5 py-1 rounded-lg text-xs font-bold transition-all ${
                  activeFilter === 'Todas'
                    ? 'bg-slate-900 dark:bg-slate-700 text-white shadow-xs'
                    : 'text-slate-600 dark:text-slate-400 hover:text-slate-900 dark:hover:text-slate-200'
                }`}
              >
                Todas
              </button>
              <button
                onClick={() => handleFilterSelect('Alta')}
                className={`px-2.5 py-1 rounded-lg text-xs font-bold flex items-center gap-1 transition-all ${
                  activeFilter === 'Alta'
                    ? 'bg-rose-600 text-white shadow-xs'
                    : 'text-rose-700 dark:text-rose-400 hover:bg-rose-50 dark:hover:bg-rose-950/40'
                }`}
              >
                <span className="w-1.5 h-1.5 rounded-full bg-rose-500" />
                Alta
              </button>
              <button
                onClick={() => handleFilterSelect('Média')}
                className={`px-2.5 py-1 rounded-lg text-xs font-bold flex items-center gap-1 transition-all ${
                  activeFilter === 'Média'
                    ? 'bg-amber-500 text-white shadow-xs'
                    : 'text-amber-800 dark:text-amber-400 hover:bg-amber-50 dark:hover:bg-amber-950/40'
                }`}
              >
                <span className="w-1.5 h-1.5 rounded-full bg-amber-500" />
                Média
              </button>
              <button
                onClick={() => handleFilterSelect('Baixa')}
                className={`px-2.5 py-1 rounded-lg text-xs font-bold flex items-center gap-1 transition-all ${
                  activeFilter === 'Baixa'
                    ? 'bg-blue-600 text-white shadow-xs'
                    : 'text-blue-700 dark:text-blue-400 hover:bg-blue-50 dark:hover:bg-blue-950/40'
                }`}
              >
                <span className="w-1.5 h-1.5 rounded-full bg-blue-500" />
                Baixa
              </button>
            </div>
          </div>

          {/* Valor Principal e Contraste com o Total Geral */}
          {activeLevelInvestment ? (
            /* VISUALIZAÇÃO QUANDO UM NÍVEL ESPECÍFICO ESTÁ SELECIONADO (ALTA, MÉDIA OU BAIXA) */
            <div className="space-y-4">
              <div className="flex flex-wrap items-baseline gap-3">
                <span className="text-3xl sm:text-4xl font-extrabold tracking-tight text-slate-900 dark:text-slate-100">
                  R$ {activeLevelInvestment.levelCost.toFixed(2).replace('.', ',')}
                </span>
                <span className="text-sm font-semibold text-slate-500 dark:text-slate-400">
                  de R$ {priorityStats.totalCost.toFixed(2).replace('.', ',')} (Total)
                </span>
                <span className={`text-xs font-bold px-2.5 py-0.5 rounded-full border ${activeLevelInvestment.badgeBg}`}>
                  {activeLevelInvestment.levelPercent}%
                </span>
              </div>

              {/* Barra Visual de Contraste: Nível Selecionado vs Restante do Total */}
              <div>
                <div className="flex justify-between text-xs font-semibold text-slate-600 dark:text-slate-300 mb-1.5">
                  <span className="flex items-center gap-1.5">
                    <span className={`w-2 h-2 rounded-full ${activeLevelInvestment.accentColor}`} />
                    {activeFilter}: R$ {activeLevelInvestment.levelCost.toFixed(2).replace('.', ',')} ({activeLevelInvestment.levelPercent}%)
                  </span>
                  <span className="text-slate-400 dark:text-slate-500">
                    Outras: R$ {activeLevelInvestment.otherCost.toFixed(2).replace('.', ',')} ({100 - activeLevelInvestment.levelPercent}%)
                  </span>
                </div>

                <div className="w-full bg-slate-100 dark:bg-slate-800 h-3 rounded-full overflow-hidden flex border border-slate-200/60 dark:border-slate-700/60">
                  <div 
                    className={`${activeLevelInvestment.accentColor} h-full transition-all duration-500`}
                    style={{ width: `${activeLevelInvestment.levelPercent}%` }}
                    title={`${activeFilter}: ${activeLevelInvestment.levelPercent}%`}
                  />
                  <div 
                    className="bg-slate-200 dark:bg-slate-700 h-full transition-all duration-500"
                    style={{ width: `${100 - activeLevelInvestment.levelPercent}%` }}
                    title={`Outras: ${100 - activeLevelInvestment.levelPercent}%`}
                  />
                </div>
              </div>

              {/* Detalhes de Quantidade e Comparativo */}
              <div className="grid grid-cols-2 sm:grid-cols-3 gap-3 pt-2 text-xs">
                <div className="bg-slate-50 dark:bg-slate-800/60 p-2.5 rounded-xl border border-slate-100 dark:border-slate-800">
                  <span className="text-slate-400 dark:text-slate-500 font-medium block">Neste Nível</span>
                  <span className="text-slate-900 dark:text-slate-100 font-bold text-sm">
                    {activeLevelInvestment.levelCount} produtos ({activeLevelInvestment.levelUnits} un.)
                  </span>
                </div>
                <div className="bg-slate-50 dark:bg-slate-800/60 p-2.5 rounded-xl border border-slate-100 dark:border-slate-800">
                  <span className="text-slate-400 dark:text-slate-500 font-medium block">Total de Itens</span>
                  <span className="text-slate-900 dark:text-slate-100 font-bold text-sm">
                    {priorityStats.totalItemsCount} produtos ({priorityStats.totalUnitsCount} un.)
                  </span>
                </div>
                <div className="bg-slate-50 dark:bg-slate-800/60 p-2.5 rounded-xl border border-slate-100 dark:border-slate-800 col-span-2 sm:col-span-1">
                  <span className="text-slate-400 dark:text-slate-500 font-medium block">Restante</span>
                  <span className="text-slate-900 dark:text-slate-100 font-bold text-sm">
                    R$ {activeLevelInvestment.otherCost.toFixed(2).replace('.', ',')}
                  </span>
                </div>
              </div>
            </div>
          ) : (
            /* VISUALIZAÇÃO QUANDO 'TODAS AS PRIORIDADES' ESTÁ SELECIONADO: CONTRASTE DE TODOS OS NÍVEIS */
            <div className="space-y-4">
              <div className="flex flex-wrap items-baseline gap-3">
                <span className="text-3xl sm:text-4xl font-extrabold tracking-tight text-slate-900 dark:text-slate-100">
                  R$ {priorityStats.totalCost.toFixed(2).replace('.', ',')}
                </span>
                <span className="text-sm font-semibold text-slate-500 dark:text-slate-400">
                  {priorityStats.totalItemsCount} produtos ({priorityStats.totalUnitsCount} un.)
                </span>
                <span className="text-xs font-bold px-2.5 py-0.5 rounded-full bg-emerald-100 dark:bg-emerald-950/60 text-emerald-800 dark:text-emerald-300 border border-emerald-200 dark:border-emerald-800/60">
                  100%
                </span>
              </div>

              {/* Barra Visual de Contraste Triplo: Alta + Média + Baixa */}
              <div>
                <div className="flex justify-between text-xs font-semibold text-slate-600 dark:text-slate-300 mb-1.5">
                  <span>Distribuição por Nível</span>
                  <span className="text-slate-400 dark:text-slate-500">Porcentagens</span>
                </div>

                <div className="w-full bg-slate-100 dark:bg-slate-800 h-3 rounded-full overflow-hidden flex border border-slate-200/60 dark:border-slate-700/60">
                  <div 
                    className="bg-rose-500 h-full transition-all duration-500"
                    style={{ width: `${priorityStats.highPercent}%` }}
                    title={`Alta: R$ ${priorityStats.highCost.toFixed(2)} (${priorityStats.highPercent}%)`}
                  />
                  <div 
                    className="bg-amber-500 h-full transition-all duration-500"
                    style={{ width: `${priorityStats.medPercent}%` }}
                    title={`Média: R$ ${priorityStats.medCost.toFixed(2)} (${priorityStats.medPercent}%)`}
                  />
                  <div 
                    className="bg-blue-500 h-full transition-all duration-500"
                    style={{ width: `${priorityStats.lowPercent}%` }}
                    title={`Baixa: R$ ${priorityStats.lowCost.toFixed(2)} (${priorityStats.lowPercent}%)`}
                  />
                </div>
              </div>

              {/* Mini-Cards de Contraste Clicáveis */}
              <div className="grid grid-cols-1 sm:grid-cols-3 gap-2.5 pt-1 text-xs">
                <button
                  onClick={() => handleFilterSelect('Alta')}
                  className="bg-rose-50/60 dark:bg-rose-950/40 hover:bg-rose-100/70 dark:hover:bg-rose-900/50 border border-rose-200 dark:border-rose-800/60 p-2.5 rounded-xl text-left transition-colors flex items-center justify-between"
                >
                  <div>
                    <div className="flex items-center gap-1.5 font-bold text-rose-800 dark:text-rose-300">
                      <span className="w-2 h-2 rounded-full bg-rose-500" /> Alta
                    </div>
                    <div className="text-slate-900 dark:text-slate-100 font-extrabold text-sm mt-0.5">
                      R$ {priorityStats.highCost.toFixed(2).replace('.', ',')}
                    </div>
                  </div>
                  <span className="text-xs font-bold text-rose-700 dark:text-rose-300 bg-white dark:bg-slate-800 px-2 py-0.5 rounded-md border border-rose-200 dark:border-rose-800/60">
                    {priorityStats.highPercent}%
                  </span>
                </button>

                <button
                  onClick={() => handleFilterSelect('Média')}
                  className="bg-amber-50/60 dark:bg-amber-950/40 hover:bg-amber-100/70 dark:hover:bg-amber-900/50 border border-amber-200 dark:border-amber-800/60 p-2.5 rounded-xl text-left transition-colors flex items-center justify-between"
                >
                  <div>
                    <div className="flex items-center gap-1.5 font-bold text-amber-900 dark:text-amber-300">
                      <span className="w-2 h-2 rounded-full bg-amber-500" /> Média
                    </div>
                    <div className="text-slate-900 dark:text-slate-100 font-extrabold text-sm mt-0.5">
                      R$ {priorityStats.medCost.toFixed(2).replace('.', ',')}
                    </div>
                  </div>
                  <span className="text-xs font-bold text-amber-800 dark:text-amber-300 bg-white dark:bg-slate-800 px-2 py-0.5 rounded-md border border-amber-200 dark:border-amber-800/60">
                    {priorityStats.medPercent}%
                  </span>
                </button>

                <button
                  onClick={() => handleFilterSelect('Baixa')}
                  className="bg-blue-50/60 dark:bg-blue-950/40 hover:bg-blue-100/70 dark:hover:bg-blue-900/50 border border-blue-200 dark:border-blue-800/60 p-2.5 rounded-xl text-left transition-colors flex items-center justify-between"
                >
                  <div>
                    <div className="flex items-center gap-1.5 font-bold text-blue-900 dark:text-blue-300">
                      <span className="w-2 h-2 rounded-full bg-blue-500" /> Baixa
                    </div>
                    <div className="text-slate-900 dark:text-slate-100 font-extrabold text-sm mt-0.5">
                      R$ {priorityStats.lowCost.toFixed(2).replace('.', ',')}
                    </div>
                  </div>
                  <span className="text-xs font-bold text-blue-800 dark:text-blue-300 bg-white dark:bg-slate-800 px-2 py-0.5 rounded-md border border-blue-200 dark:border-blue-800/60">
                    {priorityStats.lowPercent}%
                  </span>
                </button>
              </div>
            </div>
          )}
        </div>

        {/* Card de Insight e Destaque Inteligente de Prioridade */}
        <div className="bg-white dark:bg-slate-900 rounded-2xl p-6 border border-slate-200 dark:border-slate-800 shadow-sm flex flex-col justify-between transition-colors">
          <div>
            <div className="flex justify-between items-start mb-3">
              <div className="flex items-center gap-2">
                <div className="bg-amber-500 p-1.5 rounded-lg text-white">
                  <Lightbulb className="h-4 w-4" />
                </div>
                <h3 className="text-xs font-bold text-slate-400 dark:text-slate-500 uppercase tracking-wider">

                    <Minus className="h-3.5 w-3.5" />
                  </button>
                  <span className="text-xs font-bold w-5 text-center text-slate-900 dark:text-slate-100">{item.qty}</span>
                  <button 
                    onClick={() => updatePriorityItemQty(item.id, 1)}
                    className="w-7 h-7 rounded-full bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 flex items-center justify-center text-slate-500 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-700 hover:text-slate-900 dark:hover:text-white transition-colors"
                  >
                    <Plus className="h-3.5 w-3.5" />
                  </button>
                  
                  <button 
                    onClick={() => removePriorityItem(item.id)}
                    className="w-7 h-7 rounded-full flex items-center justify-center text-slate-400 hover:bg-rose-50 dark:hover:bg-rose-950/50 hover:text-rose-500 dark:hover:text-rose-400 transition-colors ml-1"
                  >
                    <Trash2 className="h-3.5 w-3.5" />
                  </button>
                </div>
              </div>
            ))
          )}
        </div>
      )}

      {/* Modal para Adicionar Novo Item */}
      {isModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-slate-900/60 backdrop-blur-xs p-4 animate-in fade-in duration-200">
          <div className="bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-800 rounded-2xl w-full max-w-md overflow-hidden shadow-xl animate-in zoom-in-95 duration-200">
            <div className="px-6 py-4 border-b border-slate-100 dark:border-slate-800 flex justify-between items-center bg-slate-50/50 dark:bg-slate-800/50">
              <h3 className="text-lg font-bold text-slate-900 dark:text-slate-100">Adicionar Item à Lista</h3>
              <button 
                onClick={() => setIsModalOpen(false)}
                className="text-slate-400 hover:text-slate-700 dark:hover:text-slate-200 transition-colors p-1 rounded-md hover:bg-slate-200 dark:hover:bg-slate-700"
              >
                <X className="w-5 h-5" />
              </button>
            </div>
            
            <div className="p-6 space-y-4">
              <div>
                <label className="block text-sm font-semibold text-slate-700 dark:text-slate-300 mb-1.5">Nome do Item</label>
                <input 
                  type="text" 
                  value={newItemName}
                  onChange={(e) => {
                    setNewItemName(e.target.value);
                    if (nameError) setNameError(false);
                  }}
                  className={`w-full px-4 py-2.5 bg-slate-50 dark:bg-slate-800 border rounded-xl focus:outline-none focus:ring-2 font-medium text-slate-900 dark:text-slate-100 placeholder:text-slate-400 dark:placeholder:text-slate-500 ${
                    nameError 
                      ? 'border-rose-400 focus:ring-rose-400' 
                      : 'border-slate-200 dark:border-slate-700 focus:ring-emerald-500'
                  }`}
                  placeholder="Ex: Purificador de Água"
                  autoFocus
                />
                {nameError && (
                  <p className="text-xs text-rose-500 dark:text-rose-400 font-medium mt-1">
                    Por favor, digite o nome do item.
                  </p>
                )}
              </div>

              <div>
                <label className="block text-sm font-semibold text-slate-700 dark:text-slate-300 mb-1.5">Categoria / Setor</label>
                <input 
                  type="text" 
                  value={newItemCategory}
                  onChange={(e) => setNewItemCategory(e.target.value)}
                  className="w-full px-4 py-2.5 bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl focus:outline-none focus:ring-2 focus:ring-emerald-500 font-medium text-slate-900 dark:text-slate-100 placeholder:text-slate-400 dark:placeholder:text-slate-500"
                  placeholder="Ex: Cozinha, Limpeza, Eletrônicos"
                />
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label className="block text-sm font-semibold text-slate-700 dark:text-slate-300 mb-1.5">Preço Estimado</label>
                  <input 
                    type="text" 
                    value={newItemPrice}
                    onChange={(e) => setNewItemPrice(e.target.value)}
                    className="w-full px-4 py-2.5 bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl focus:outline-none focus:ring-2 focus:ring-emerald-500 font-medium text-slate-900 dark:text-slate-100 placeholder:text-slate-400 dark:placeholder:text-slate-500"
                    placeholder="Ex: 150,00"
                  />
                </div>
                <div>
                  <label className="block text-sm font-semibold text-slate-700 dark:text-slate-300 mb-1.5">Prioridade</label>
                  <div className="flex gap-2">
                    {(['Alta', 'Média', 'Baixa'] as const).map((p) => (
                      <button
                        key={p}
                        onClick={() => setNewItemPriority(p)}
                        className={`flex-1 flex items-center justify-center gap-2 text-[11px] font-bold uppercase tracking-wider py-2.5 rounded-xl transition-all cursor-pointer border ${
                          newItemPriority === p 
                            ? p === 'Alta' 
                              ? 'bg-rose-600 text-white border-rose-600 shadow-sm' 
                              : p === 'Média'
                              ? 'bg-amber-500 text-white border-amber-500 shadow-sm'
                              : 'bg-blue-600 text-white border-blue-600 shadow-sm'
                            : 'bg-white dark:bg-slate-800 border-slate-200 dark:border-slate-700 text-slate-600 dark:text-slate-400 hover:bg-slate-50 dark:hover:bg-slate-750'
                        }`}
                      >
                        <span className={`w-2 h-2 rounded-full ${
                          newItemPriority === p 
                            ? 'bg-white' 
                            : p === 'Alta' ? 'bg-rose-500' : p === 'Média' ? 'bg-amber-500' : 'bg-blue-500'
                        }`} />
                        {p}
                      </button>
                    ))}
                  </div>
                </div>
              </div>
            </div>

            <div className="px-6 py-4 border-t border-slate-100 dark:border-slate-800 bg-slate-50 dark:bg-slate-800/50 flex justify-end gap-3">
              <button 
                onClick={() => setIsModalOpen(false)}
                className="px-4 py-2 text-sm font-semibold text-slate-600 dark:text-slate-400 hover:bg-slate-200 dark:hover:bg-slate-700 rounded-xl transition-colors cursor-pointer"
              >
                Cancelar
              </button>
              <button 
                onClick={handleAdd}
                className="px-6 py-2 text-sm font-semibold text-white bg-slate-900 dark:bg-emerald-600 hover:bg-slate-800 dark:hover:bg-emerald-500 active:scale-95 rounded-xl transition-all shadow-sm cursor-pointer"
              >
                Adicionar à Lista
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
